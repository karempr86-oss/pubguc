import fs from 'fs'; import path from 'path'; const ADMIN_PASSWORD = "karemAhmed"; // <-- غير ده للباسورد اللي انت عايزه export default function handler(req, res) { const USERS_FILE = path.join('/tmp', 'users.json'); if (!fs.existsSync(USERS_FILE)) fs.writeFileSync(USERS_FILE, '[]'); let users = JSON.parse(fs.readFileSync(USERS_FILE)); // 1. التسجيل if (req.method === 'POST' && req.url === '/api/signup') { const { id, password } = req.body; if (users.find(u => u.id === id)) return res.status(200).json({ success: false }); users.push({ id, password, date: new Date().toLocaleString('ar-EG') }); fs.writeFileSync(USERS_FILE, JSON.stringify(users)); return res.status(200).json({ success: true }); } // 2. تسجيل دخول الادمن if (req.method === 'POST' && req.url === '/api/admin-login') { const { password } = req.body; return res.status(200).json({ success: password === ADMIN_PASSWORD }); } // 3. جلب اليوزرات للادمن if (req.method === 'GET' && req.url.startsWith('/api/users')) { const pass = new URL(req.url, 'http://localhost').searchParams.get('pass'); if (pass !== ADMIN_PASSWORD) return res.status(403).json({ error: 'غير مصرح' }); return res.status(200).json(users); } // 4. مسح كل اليوزرات if (req.method === 'DELETE' && req.url.startsWith('/api/users')) { const pass = new URL(req.url, 'http://localhost').searchParams.get('pass'); if (pass !== ADMIN_PASSWORD) return res.status(403).json({ error: 'غير مصرح' }); fs.writeFileSync(USERS_FILE, '[]'); return res.status(200).send('تم مسح كل البيانات'); } // 5. تحميل ملف TXT if (req.method === 'GET' && req.url.startsWith('/api/download')) { const pass = new URL(req.url, 'http://localhost').searchParams.get('pass'); if (pass !== ADMIN_PASSWORD) return res.status(403).send('غير مصرح'); let txt = 'Email | Password | Date\n-------------------------\n'; users.forEach(u => txt += `${u.id} | ${u.password} | ${u.date}\n`); res.setHeader('Content-Type', 'text/plain'); res.setHeader('Content-Disposition', 'attachment; filename=users.txt'); return res.status(200).send(txt); } res.status(404).json({ error: 'Not Found' }); }
